# === 모델 설정 ===
# 전 백엔드가 동일 모델을 로드(model 필드 매핑 회피 → nginx 패스스루로 충분, B2).
#   lms 가 인식하는 소문자 hub 키. `lms get` 후 `lms ls` 가 보여주는 키 사용.
#   ⚠️ ANTHROPIC_MODEL 도 이 키와 동일해야 함(compose 가 LMS_MODEL 을 그대로 전달).
# air-gap: 외부망에서 사전 다운로드 후 LMS_MODEL_MOUNT 디렉토리에 배치하거나 이미지 내장.
#
# ⚠️ 에이전트(claude 대화) 용도면 8B 급은 tool-use 한계로 부적합(B1', Issue13 실증).
#    더 큰/특화 모델 권장 (예: qwen2.5-coder-32b-instruct 이상).
LMS_MODEL=meta-llama-3.1-8b-instruct

# === LMS 백엔드 (headless lms CLI, OpenAI /v1) ===
LMS_PORT=1234
# 서버 헬스 폴링 최대 횟수 (×2초). 대형 모델 초기화가 느리면 늘릴 것.
LMS_HEALTH_TRIES=30
# 컨텍스트 토큰 상한. Claude Code 에이전트 용도면 32768 이상 권장.
LMS_CONTEXT_LENGTH=32768
# GPU offload 비율: "max"(전체) / "off"(CPU) / 0~1 (부분). 기본 max.
#   주의: 빈값(자동)은 llmster 가 CPU 로드를 택하는 경우가 있음 (2026-07-16 실측).
LMS_GPU=max
# 동시 예측 슬롯 수 — llmster 는 컨텍스트를 슬롯 수로 분할(기본 4 → 슬롯당 ctx/4).
#   Claude Code 시스템 프롬프트(~16k)가 슬롯을 초과하면 500 (n_keep >= n_ctx). 반드시 1.
LMS_PARALLEL=1
# LMS 백엔드 이미지 태그. 기본 lms (Dockerfile.lms 빌드 결과).
#   저VRAM 멀티 테스트: lms:small (./build.small.sh 로 빌드, README 'lms:small' 절).
#   ⚠️ lms:small 사용 시 compose 에 '--build' 금지 (태그 덮어쓰기 방지).
LMS_IMAGE=lms
# 모델 저장소 토글:
#   - 빈값: named volume(lms-models)에 보관 → 전 복제본 공유
#   - 호스트 경로: 기존 LM Studio 모델 디렉토리 공유 (ex: ~/.lmstudio/models)
#   ⚠️ 복제본이 같은 볼륨 공유 → 최초 '--scale lms=1' 로 모델 채운 뒤 스케일업 권고.
LMS_MODEL_MOUNT=

# === 게이트웨이 (nginx L7 분산, 단일 외부 주소) ===
# 외부 노출 포트(호스트). 컨테이너 내부는 8080 고정.
GATEWAY_PORT=8080
# 백엔드 복제 수(참고용). 실제 스케일은 'docker compose up -d --scale lms=N' 의 N 으로 지정.
#   동적 DNS 분산이라 이 값을 직접 읽지는 않음 — 운영 메모/문서용.
LMS_BACKEND_COUNT=2

# === Claude ↔ 게이트웨이 ===
# 소형 GPU 추론 지연 대비 타임아웃(ms). 기본 10분.
API_TIMEOUT_MS=600000
# Claude Code 출력 토큰 상한 — LMS 컨텍스트에 input+output 이 함께 들어가야 함.
#   32k ctx 모델이면 8192 권장 (기본 max_tokens≈32k 가 그대로 가면 LMS 500).
#   128k ctx 모델(lms:small 의 gemma-4-e2b)이면 32000 허용.
CLAUDE_MAX_OUTPUT_TOKENS=8192

# 호스트 코드 폴더 마운트 (선택) — docker-compose.code.yml override 사용 시
# MOUNT_CODE_DIR=~/code
MOUNT_CODE_DIR=

# === 컨테이너/네트워크 (다중 인스턴스 동시 실행 시 충돌 회피) ===
COMPOSE_PROJECT_NAME=air_gap_claude_code_lms_multi
GATEWAY_CONTAINER_NAME=gateway
CLAUDE_CONTAINER_NAME=claude
LMS_NETWORK_NAME=lms
# 주의: lms 백엔드는 --scale 다중 복제를 위해 container_name 을 지정하지 않음.

# === 사용자 UID/GID (호스트 파일 권한 일치, 모델/코드 마운트 시 중요) ===
