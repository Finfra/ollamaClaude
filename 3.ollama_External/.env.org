# 모델 설정
# 사용할 Ollama 모델 태그 — 외부(호스트) Ollama 에 이미 pull 되어 있어야 함
# (이 토폴로지는 컨테이너가 모델을 pull 하지 않음. 호스트에서 `ollama pull <model>` 선행)
OLLAMA_MODEL=gemma4:26b

# === 외부 Ollama 연결 (이 토폴로지의 핵심) ===
# 호스트/서버에 이미 설치된 Ollama(systemd 등)에 claude 컨테이너만 붙음
#   - 같은 머신의 호스트 Ollama: OLLAMA_HOST=host.docker.internal (기본, extra_hosts 로 해석)
#   - 원격 서버 Ollama: OLLAMA_HOST=192.168.0.10 등 실제 IP/호스트명
#   - 호스트 Ollama 가 0.0.0.0 바인드여야 컨테이너에서 접근 가능
#     (systemd: `OLLAMA_HOST=0.0.0.0` 환경변수 설정 후 재시작)
OLLAMA_HOST=host.docker.internal
OLLAMA_PORT_EXT=11434

# 호스트 코드 폴더 마운트 (선택)
# 사용 시 docker-compose.code.yml override 적용
# MOUNT_CODE_DIR=~/code
MOUNT_CODE_DIR=

# === 컨테이너 (다중 인스턴스 동시 실행 시 충돌 회피) ===
COMPOSE_PROJECT_NAME=air_gap_claude_code_external
CLAUDE_CONTAINER_NAME=claude

# === 사용자 UID/GID (호스트 파일 권한 일치, 코드 마운트 시 중요) ===
# Linux 호스트: `id -u`, `id -g` 결과로 맞출 것 권장
# macOS Docker Desktop: 1000 그대로 두어도 자동 위임됨
USER_UID=1000
USER_GID=1000

# === 타임존 (컨테이너 내 git commit·로그 시각) ===
TZ=Asia/Seoul

#   이 토폴로지의 컨테이너는 Ollama 를 구동하지 않으므로 OLLAMA_FLASH_ATTENTION/
#   KV_CACHE_TYPE/NUM_GPU/CONTEXT_LENGTH 변수가 불필요함 (호스트 Ollama 에서 설정).
