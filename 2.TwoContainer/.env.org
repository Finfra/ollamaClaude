OLLAMA_MODEL=gemma4:26b

# 호스트 코드 폴더 마운트 (선택)

MOUNT_CODE_DIR=

# === 컨테이너·네트워크·포트 (다중 인스턴스 동시 실행 시 충돌 회피) ===

# Linux 호스트: `id -u`, `id -g` 결과로 맞출 것 권장
# macOS Docker Desktop: 1000 그대로 두어도 자동 위임됨

OLLAMA_NUM_GPU=999
# 컨텍스트 길이
